## 1. ORM FUNDAMENTALS

### Model Definition
```python
from odoo import models, fields, api

class MyModel(models.Model):
    _name = 'my.model'
    _description = 'My Model Description'
    _order = 'name asc, id desc'
    _rec_name = 'name'  # Field used for display_name
    
    # Fields
    name = fields.Char(string='Name', required=True)
    active = fields.Boolean(default=True)  # Enables archive feature
    company_id = fields.Many2one('res.company', default=lambda self: self.env.company)
```

### Field Types
```python
# Basic Fields
name = fields.Char(string='Name', size=100, required=True, index=True, translate=True)
description = fields.Text(string='Description')

    ('draft', 'Draft'),
    ('confirmed', 'Confirmed'),

# Relational Fields
partner_id = fields.Many2one('res.partner', string='Partner', ondelete='cascade')
# ondelete options: 'cascade', 'set null', 'restrict'

line_ids = fields.One2many('my.model.line', 'parent_id', string='Lines')

# Full syntax: fields.Many2many('my.tag', 'my_model_tag_rel', 'model_id', 'tag_id')

```

### Computed Fields
```python
total = fields.Float(string='Total', compute='_compute_total', store=True)

@api.depends('line_ids.amount', 'line_ids.quantity')
def _compute_total(self):
    for record in self:
        record.total = sum(line.amount * line.quantity for line in record.line_ids)

# With inverse (allows user to set value)
date_deadline = fields.Date(compute='_compute_deadline', inverse='_inverse_deadline', store=True)

@api.depends('create_date', 'validity')
def _compute_deadline(self):
    for record in self:
        if record.create_date:
            record.date_deadline = record.create_date + timedelta(days=record.validity)

def _inverse_deadline(self):
    for record in self:
        if record.date_deadline and record.create_date:
            record.validity = (record.date_deadline - record.create_date.date()).days
```

### Odoo 19 Computed Field Enhancements
```python
# Precompute for faster batch creation
total = fields.Float(compute='_compute_total', store=True, precompute=True)

# Recursive dependencies (improved in 19)
@api.depends('parent_id.accumulated_value')
def _compute_accumulated(self):
    for record in self:
        record.accumulated_value = record.value + (record.parent_id.accumulated_value or 0)

accumulated_value = fields.Float(
    compute='_compute_accumulated', 
    store=True, 
    recursive=True  # Required for recursive dependencies
)

# compute_sudo for bypassing access rights during computation
sensitive_total = fields.Float(
    compute='_compute_sensitive', 
    store=True, 
    compute_sudo=True  # Runs as superuser
)
```

### Model Inheritance

```python
class ResPartner(models.Model):
    _inherit = 'res.partner'  # Extends existing model
```

### Constraints
```python
from odoo.exceptions import ValidationError

# SQL Constraint
_sql_constraints = [
    ('name_unique', 'UNIQUE(name)', 'Name must be unique!'),
    ('check_amount', 'CHECK(amount >= 0)', 'Amount must be positive!'),
]

# Python Constraint
@api.constrains('date_start', 'date_end')
def _check_dates(self):
    for record in self:
        if record.date_start and record.date_end:
            if record.date_start > record.date_end:
                raise ValidationError("End date must be after start date!")
```

### Basic Structure
```xml

    <field name="model">my.model</field>
    <field name="inherit_id" ref="module_name.view_my_model_form"/>
    <field name="arch" type="xml">
        <!-- inheritance specs here -->
    </field>
</record>
```

### XPath Expressions
```xml
<!-- Find by field name -->
<xpath expr="//field[@name='partner_id']" position="after">
    <field name="new_field"/>
</xpath>

<!-- Find by element name attribute -->
<xpath expr="//page[@name='other_info']" position="inside">
    <field name="custom_field"/>
</xpath>

<!-- Find by position in tree -->
<xpath expr="//sheet/group[1]" position="after">
    <group name="custom_group">
        <field name="field1"/>
    </group>
</xpath>

<!-- hasclass() function for QWeb views -->
<xpath expr="//div[hasclass('o_form_sheet')]" position="inside">
    <field name="custom"/>
</xpath>

<!-- Shorthand (when element is unique) -->
<field name="partner_id" position="after">
    <field name="new_field"/>
</field>
```

### Position Attributes
```xml
<!-- before: Add before the matched element -->
<field name="name" position="before">
    <field name="code"/>
</field>

<!-- after: Add after the matched element -->
<field name="name" position="after">
    <field name="description"/>
</field>

<!-- inside: Add inside the matched element (at end) -->
<xpath expr="//page[@name='details']" position="inside">
    <field name="extra_info"/>
</xpath>

<!-- replace: Replace the matched element entirely -->
<field name="old_field" position="replace">
    <field name="new_field"/>
</field>

<!-- replace with $0 to keep original -->
<field name="name" position="replace">
    <div class="custom_wrapper">
        $0  <!-- Original element inserted here -->
    </div>
</field>

<!-- attributes: Modify attributes -->
<field name="partner_id" position="attributes">
    <attribute name="invisible">1</attribute>
    <attribute name="required">1</attribute>
    <attribute name="readonly">state != 'draft'</attribute>
    <attribute name="domain">[('is_company', '=', True)]</attribute>
</field>

<!-- move: Move element to new location -->
<xpath expr="//field[@name='target_field']" position="after">
    <xpath expr="//field[@name='field_to_move']" position="move"/>
</xpath>
```

### View Modes
```xml
<!-- Primary mode: Creates independent view based on parent -->
<record id="view_custom_form" model="ir.ui.view">
    <field name="name">custom.form</field>
    <field name="model">res.partner</field>
```

### Recompute Stored Computed Fields (Odoo 19)
```python
# Recommended method for Odoo 17+
model = env['my.model']
records = model.search([])

# Method 1: Using add_to_compute
env.add_to_compute(model._fields['computed_field'], records)
model.recompute()
env.cr.commit()

# Method 2: Direct method call (simplest)
for record in records:
    record._compute_my_field()
env.cr.commit()

# Method 3: Using _recompute_recordset
env.add_to_compute(model._fields['computed_field'], records)
model._recompute_recordset()
env.cr.commit()

# Batch processing for large datasets
batch_size = 1000
all_records = env['my.model'].search([])
for i in range(0, len(all_records), batch_size):
    batch = all_records[i:i + batch_size]
    for record in batch:
        record._compute_my_field()
    env.cr.commit()
    print(f"Processed {min(i + batch_size, len(all_records))} / {len(all_records)}")
```

## 6. COMMON PATTERNS

### Sequence Generation
```python
class MyModel(models.Model):
    _name = 'my.model'
    
    name = fields.Char(string='Reference', readonly=True, default='New', copy=False)
    
    @api.model_create_multi
    def create(self, vals_list):
        for vals in vals_list:
            if vals.get('name', 'New') == 'New':
                vals['name'] = self.env['ir.sequence'].next_by_code('my.model') or 'New'
        return super().create(vals_list)
```

### State Machine
```python
class MyModel(models.Model):
    _name = 'my.model'
    
    state = fields.Selection([
        ('draft', 'Draft'),
        ('confirmed', 'Confirmed'),
        ('done', 'Done'),
        ('cancelled', 'Cancelled'),
    ], default='draft', tracking=True, copy=False)
    
    def action_confirm(self):
        self.ensure_one()
        if self.state != 'draft':
            raise UserError("Only draft orders can be confirmed!")
        self.write({'state': 'confirmed'})
    
    def action_done(self):
        self.write({'state': 'done'})
    
    def action_cancel(self):
        self.write({'state': 'cancelled'})
    
    def action_draft(self):
        self.write({'state': 'draft'})
```

## 9. MIGRATION SCRIPTS

### Pre-migrate (before module update)
```python
# migrations/19.0.1.0.1/pre-migrate.py
def migrate(cr, version):
    # Rename column before ORM sees it
    cr.execute("""
        ALTER TABLE my_model 
        RENAME COLUMN old_name TO new_name
    """)
    
    # Add column if missing
    cr.execute("""
        ALTER TABLE my_model 
        ADD COLUMN IF NOT EXISTS new_field VARCHAR
    """)
```

### Post-migrate (after module update)
```python
# migrations/19.0.1.0.1/post-migrate.py
from odoo import api, SUPERUSER_ID

def migrate(cr, version):
    env = api.Environment(cr, SUPERUSER_ID, {})
    
    # Update data using ORM
    records = env['my.model'].search([('state', '=', 'old_state')])
    records.write({'state': 'new_state'})
    
    # Recompute fields
    model = env['my.model']
    env.add_to_compute(model._fields['computed_field'], model.search([]))
    model.recompute()
    
    # Or direct SQL for performance
    cr.execute("""
        UPDATE my_model 
        SET state = 'new_state' 
        WHERE state = 'old_state'
    """)
```

### Using openupgradelib (recommended for complex migrations)
```python
from openupgradelib import openupgrade

@openupgrade.migrate()
def migrate(env, version):
    openupgrade.rename_fields(env, [
        ('my.model', 'my_model', 'old_field', 'new_field'),
    ])
    openupgrade.rename_models(env.cr, [
        ('old.model', 'new.model'),
    ])
```

```javascript
import { Component } from "@odoo/owl";

class MyComponent extends Component {
    static template = "my_module.MyComponent";
    static props = {
        record: Object,
    };

    }
```

```bash
# Start server
odoo-bin -c odoo.conf

odoo-bin -c odoo.conf -d database -u module_name

# Install module  
odoo-bin -c odoo.conf -d database -i module_name

odoo-bin shell -c odoo.conf -d database

# Database operations
odoo-bin -c odoo.conf -d database --stop-after-init  # Test startup
```

```sql
FROM ir_module_module 
WHERE name LIKE '%my_module%';
```

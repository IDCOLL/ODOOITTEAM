## 1. ORM FUNDAMENTALS

### Model Definition
```python
from odoo import models, fields, api

class MyModel(models.Model):
    _name = 'my.model'
    _description = 'My Model Description'
    _order = 'name asc, id desc'
    _rec_name = 'name'  # Field used for display_name
    
    # Fields
    name = fields.Char(string='Name', required=True)
    active = fields.Boolean(default=True)  # Enables archive feature
    company_id = fields.Many2one('res.company', default=lambda self: self.env.company)
```

### Field Types
```python
# Basic Fields
name = fields.Char(string='Name', size=100, required=True, index=True, translate=True)
description = fields.Text(string='Description')
amount = fields.Float(string='Amount', digits=(16, 2))
quantity = fields.Integer(string='Quantity', default=1)
is_active = fields.Boolean(string='Active', default=True)
date_start = fields.Date(string='Start Date', default=fields.Date.today)
datetime_field = fields.Datetime(string='Datetime', default=fields.Datetime.now)
state = fields.Selection([
    ('draft', 'Draft'),
    ('confirmed', 'Confirmed'),
    ('done', 'Done'),
], string='Status', default='draft', required=True)
html_content = fields.Html(string='Content', sanitize=True)
binary_file = fields.Binary(string='File', attachment=True)
image = fields.Image(string='Image', max_width=1024, max_height=1024)

# Relational Fields
partner_id = fields.Many2one('res.partner', string='Partner', ondelete='cascade')
# ondelete options: 'cascade', 'set null', 'restrict'

line_ids = fields.One2many('my.model.line', 'parent_id', string='Lines')
tag_ids = fields.Many2many('my.tag', string='Tags')
# Full syntax: fields.Many2many('my.tag', 'my_model_tag_rel', 'model_id', 'tag_id')

# Related/Computed
partner_name = fields.Char(related='partner_id.name', string='Partner Name', store=True)
```

### Computed Fields
```python
total = fields.Float(string='Total', compute='_compute_total', store=True)

@api.depends('line_ids.amount', 'line_ids.quantity')
def _compute_total(self):
    for record in self:
        record.total = sum(line.amount * line.quantity for line in record.line_ids)

# With inverse (allows user to set value)
date_deadline = fields.Date(compute='_compute_deadline', inverse='_inverse_deadline', store=True)

@api.depends('create_date', 'validity')
def _compute_deadline(self):
    for record in self:
        if record.create_date:
            record.date_deadline = record.create_date + timedelta(days=record.validity)

def _inverse_deadline(self):
    for record in self:
        if record.date_deadline and record.create_date:
            record.validity = (record.date_deadline - record.create_date.date()).days
```

### Model Inheritance

#### Extension Inheritance (most common)
```python
class ResPartner(models.Model):
    _inherit = 'res.partner'  # Extends existing model
    
    custom_field = fields.Char(string='Custom Field')
    
    def action_custom(self):
        # Add new method
        pass
    
    def write(self, vals):
        # Override existing method
        res = super().write(vals)
        # Custom logic after write
        return res
```

#### Delegation Inheritance
```python
class ChildModel(models.Model):
    _name = 'child.model'
    _inherits = {'parent.model': 'parent_id'}  # Delegation
    
    parent_id = fields.Many2one('parent.model', required=True, ondelete='cascade')
    child_field = fields.Char()
```

### CRUD Operations
```python
# Create
record = self.env['my.model'].create({
    'name': 'Test',
    'partner_id': partner.id,

# Update

record.name = 'New Name'  # Direct assignment

# Delete
record.unlink()

# One2many/Many2many special commands
# (0, 0, values) - Create new record
# (1, id, values) - Update existing record
# (2, id, 0) - Delete record
# (3, id, 0) - Unlink (remove from relation, don't delete)
# (4, id, 0) - Link existing record
# (5, 0, 0) - Unlink all
# (6, 0, [ids]) - Replace all with provided ids
```

### Search and Domains
```python
# Basic search
records = self.env['res.partner'].search([
    ('is_company', '=', True),
    ('country_id.code', '=', 'US'),
])

# Domain operators
# =, !=, <, >, <=, >=
# in, not in
# like, ilike (case insensitive), =like, =ilike
# child_of, parent_of (hierarchical)

# Logic operators
# '&' (AND, default), '|' (OR), '!' (NOT)
domain = ['|', ('state', '=', 'draft'), ('state', '=', 'sent')]

# Search options
records = self.search(domain, limit=10, offset=0, order='name asc')
count = self.search_count(domain)

# Filtered/Mapped/Sorted
active_records = records.filtered(lambda r: r.active)
active_records = records.filtered('active')  # Shorthand
names = records.mapped('name')
partner_names = records.mapped('partner_id.name')
sorted_records = records.sorted(key='name', reverse=True)
```

### Constraints
```python
from odoo.exceptions import ValidationError

# SQL Constraint
_sql_constraints = [
    ('name_unique', 'UNIQUE(name)', 'Name must be unique!'),
    ('check_amount', 'CHECK(amount >= 0)', 'Amount must be positive!'),
]

# Python Constraint
@api.constrains('date_start', 'date_end')
def _check_dates(self):
    for record in self:
        if record.date_start and record.date_end:
            if record.date_start > record.date_end:
                raise ValidationError("End date must be after start date!")
```

### Onchange Methods

        }}

## 2. VIEW INHERITANCE

### Basic Structure
```xml
<record id="view_my_model_form_inherit" model="ir.ui.view">
    <field name="name">my.model.form.inherit</field>
    <field name="model">my.model</field>
    <field name="inherit_id" ref="module_name.view_my_model_form"/>
    <field name="arch" type="xml">
        <!-- inheritance specs here -->
    </field>
</record>
```

### XPath Expressions
```xml
<!-- Find by field name -->
<xpath expr="//field[@name='partner_id']" position="after">
    <field name="new_field"/>
</xpath>

<!-- Find by element name attribute -->
<xpath expr="//page[@name='other_info']" position="inside">
    <field name="custom_field"/>
</xpath>

<!-- Find by position in tree -->
<xpath expr="//sheet/group[1]" position="after">
    <group name="custom_group">
        <field name="field1"/>
    </group>
</xpath>

<!-- Shorthand (when element is unique) -->
<field name="partner_id" position="after">
    <field name="new_field"/>
</field>
```

### Position Attributes
```xml
<!-- before: Add before the matched element -->
<field name="name" position="before">
    <field name="code"/>
</field>

<!-- after: Add after the matched element -->
<field name="name" position="after">
    <field name="description"/>
</field>

<!-- inside: Add inside the matched element (at end) -->
<xpath expr="//page[@name='details']" position="inside">
    <field name="extra_info"/>
</xpath>

<!-- replace: Replace the matched element entirely -->
<field name="old_field" position="replace">
    <field name="new_field"/>
</field>

<!-- attributes: Modify attributes -->
<field name="partner_id" position="attributes">
    <attribute name="invisible">1</attribute>
    <attribute name="required">1</attribute>
    <attribute name="readonly">state != 'draft'</attribute>
    <attribute name="domain">[('is_company', '=', True)]</attribute>
</field>

<!-- move: Move element to new location -->
<xpath expr="//field[@name='target_field']" position="after">
    <xpath expr="//field[@name='field_to_move']" position="move"/>
</xpath>
```

### View Modes
```xml
<!-- Primary mode: Creates independent view based on parent -->
<record id="view_custom_form" model="ir.ui.view">
    <field name="name">custom.form</field>
    <field name="model">res.partner</field>
    <field name="inherit_id" ref="base.view_partner_form"/>
    <field name="mode">primary</field>  <!-- Key difference -->
    <field name="arch" type="xml">
        <xpath expr="//field[@name='name']" position="after">
            <field name="custom"/>
        </xpath>
    </field>
</record>
```

### Recompute Stored Computed Fields
```python
# Odoo 13+ method (recommended)
model = env['my.model']
env.add_to_compute(model._fields['computed_field'], model.search([]))
model.recompute()
env.cr.commit()

# Alternative: Direct method call (Odoo 17+)
for record in env['my.model'].search([]):
    record._compute_my_field()
env.cr.commit()

# Batch processing for large datasets
batch_size = 1000
records = env['my.model'].search([])
for i in range(0, len(records), batch_size):
    batch = records[i:i + batch_size]
    env.add_to_compute(model._fields['computed_field'], batch)
    model.recompute()
    env.cr.commit()
```

### Common Data Fixes
```python
# Mass update
env['res.partner'].search([('country_id', '=', False)]).write({

})
env.cr.commit()

# Fix orphaned records
orphans = env['sale.order.line'].search([('order_id', '=', False)])

# Reset sequences
env['ir.sequence'].search([('code', '=', 'sale.order')]).write({

# Direct SQL (use sparingly)
env.cr.execute("""
    UPDATE res_partner

```

## 6. COMMON PATTERNS

### Sequence Generation
```python
class MyModel(models.Model):
    _name = 'my.model'
    
    name = fields.Char(string='Reference', readonly=True, default='New')
    
    @api.model_create_multi
    def create(self, vals_list):
        for vals in vals_list:
            if vals.get('name', 'New') == 'New':
                vals['name'] = self.env['ir.sequence'].next_by_code('my.model') or 'New'
        return super().create(vals_list)
```

### State Machine
```python
class MyModel(models.Model):
    _name = 'my.model'
    
    state = fields.Selection([
        ('draft', 'Draft'),
        ('confirmed', 'Confirmed'),
        ('done', 'Done'),
        ('cancelled', 'Cancelled'),
    ], default='draft', tracking=True)
    
    def action_confirm(self):
        self.write({'state': 'confirmed'})
    
    def action_done(self):
        self.write({'state': 'done'})
    
    def action_cancel(self):
        self.write({'state': 'cancelled'})
    
    def action_draft(self):
        self.write({'state': 'draft'})
```

partners = self.env['res.partner'].search([])

## 9. MIGRATION SCRIPTS

### Pre-migrate (before module update)
```python
# migrations/18.0.1.0.1/pre-migrate.py
def migrate(cr, version):
    # Rename column before ORM sees it
    cr.execute("""
        ALTER TABLE my_model 
        RENAME COLUMN old_name TO new_name
    """)
```

### Post-migrate (after module update)
```python
# migrations/18.0.1.0.1/post-migrate.py
from odoo import api, SUPERUSER_ID

def migrate(cr, version):
    env = api.Environment(cr, SUPERUSER_ID, {})
    
    # Update data using ORM
    records = env['my.model'].search([('state', '=', 'old_state')])
    records.write({'state': 'new_state'})
    
    # Or direct SQL
    cr.execute("""
        UPDATE my_model 
        SET state = 'new_state' 
        WHERE state = 'old_state'
    """)
```

def _compute_total(self):
